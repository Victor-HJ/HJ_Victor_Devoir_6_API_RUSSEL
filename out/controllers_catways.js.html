/**

 * @module controllers/catways

const Catway = require('../models/catway');

/**
 * Get every catway and sorts them in an decreasing numerical order (with catwayNumber)
 * 
 * @async
 * @function getAllCatways
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Array} 200 : An array of all registered catways
 * @throws {Error} 404 : If the array is empty | 500 : If the server encoutered an error
 */

exports.getAllCatways = async (req, res, next) => {

    try{

        const allCatways = await Catway.find();

        allCatways.sort((a, b) => b.catwayNumber - a.catwayNumber); 

        if(allCatways) {
            
            return res.status(200).json(allCatways);

        }

        if(allCatways.length === 0){

            return res.status(404).json('Aucun catway trouvé'); /* Avoids an empty array if no catway is registered */

        }

    } catch (error) {

        return res.status(500).json('Erreur serveur lors de la récupération des catways');
        
    }
}

/**
 * Get one specific catway found by its number
 * 
 * @async
 * @function getCatwayById
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Object} 200 : The catway that matches the requested number
 * @returns {Error} 404 | 500
 */

exports.getCatwayById = async (req, res, next) => {

    const id = req.params.id

    try {

        let catway = await Catway.findOne({catwayNumber : parseInt(id)});

        if(catway) {

            return res.status(200).json(catway);
        }

        return res.status(404).json('Aucun catway avec ce numéro trouvé');

    } catch (error) {

        return res.status(500).json('Erreur serveur lors de la récupération de ce catway');

    }
}

/**
 * CREATES A CATWAY <br><br>
 * 
 * - Check if mandatory fields are included in the request and are not empty (catwayNumber | catwayType | catwayState) (400) <br>
 * - Check if catwayType is valid : either 'long' or 'short' (400) <br>
 * - Check if the requested number isn't already associated to an existing catway (409)
 * 
 * @async
 * @function createCatway
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Object} 201 : The created catway
 * @returns {Error} 400 | 409 | 500
 */

exports.createCatway = async (req, res, next) => {

    const {catwayNumber, catwayType, catwayState} = req.body;
  
    try {

        if (catwayNumber === undefined || catwayType === undefined || catwayState === undefined) {

            return res.status(400).json('Tous les champs sont requis');
        }

        if(catwayNumber.trim() === '' || catwayType.trim() === '' || catwayState.trim() === '') {

            return res.status(400).json('Aucun champ ne peut être vide');

        }

        if (catwayType !== 'short' && catwayType !== 'long') {

            return res.status(400).json('Un catway ne peut être que de type "long" ou "short"');

        } 

        const preExistingCatway = await Catway.findOne({catwayNumber});

        if(preExistingCatway) {

            return res.status(409).json('Un catway est déjà associé à ce numéro')

        } else {

            const newCatway = new Catway({
                catwayNumber,
                catwayType,
                catwayState
            });

            await newCatway.save();

            return res.status(201).json(newCatway);
        }

    } catch(error) {

        return res.status(500).json('Erreur serveur lors de la création du catway');

    }
}

/**
 * UPDATES A CATWAY <br><br>
 * 
 * - Check if catwayState's field is sent and if it's not empty (400) <br>
 * - Check if catway to update exists (404) <br>
 * - Check if catwayNumber or catwayType is being modified (400)
 * 
 * @async
 * @function updateCatway
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Object} 201 : The updated catway
 * @returns {Error} 400 | 404 | 500
 */

exports.updateCatway = async (req, res, next) => {

    const {catwayState, catwayNumber, catwayType} = req.body;
    const id = req.params.id

    try {

        
        if(catwayState === undefined || catwayState.trim() === '') {

        return res.status(400).json("L'état du catway doit être renseigné et celui-ci ne peut être vide");

        }

        const updatedCatway = await Catway.findOne({catwayNumber : parseInt(id)});

        if(!updatedCatway) {

            return res.status(404).json('Catway non trouvé');

        } else {

            if (updatedCatway.catwayNumber !== parseInt(catwayNumber) || updatedCatway.catwayType !==catwayType) {

                return res.status(400).json("Seul l'état du catway peut être modifié")

            } else {

                updatedCatway.catwayState = catwayState;

                await updatedCatway.save();

                return res.status(201).json(updatedCatway);
            }
        }

    } catch (error) {

        return res.status(500).json('Erreur serveur lors de la modification du catway');

    }
}

/**
 * DELETE A CATWAY <br><br>
 * 
 * - Check if catway exists (404)
 * 
 * @async
 * @function deleteCatway
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Response} 200 : A message notifying of the successful catway deletion
 * @returns {Error} 404 | 500
 */


exports.deleteCatway = async (req, res, next) => {

    const id = req.params.id;

    try {

        const catwayToDelete = await Catway.findOne({catwayNumber : parseInt(id)});

        if(!catwayToDelete) {

            return res.status(404).json("Le catway que vous voulez supprimer n'existe pas");

        } else {

            await catwayToDelete.deleteOne();

            return res.status(200).json('Le catway a été supprimé'); /** 200 allows a message, 204 doesn't */

        }

    } catch(error) {

        return res.status(500).json('Erreur serveur lors de la suppression du catway');

    }
}
