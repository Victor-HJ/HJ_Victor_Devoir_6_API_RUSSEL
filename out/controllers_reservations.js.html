/**
 * @fileoverview Reservations controller
 * @module controllers/reservations
 */

const Reservation = require('../models/reservation');
const Catway = require('../models/catway');

/**
 * GET ALL RESERVATIONS <br><br>
 * 
 * Fetch all reservations from the database and sort them by catway number (in an increasing order) <br><br>
 * 
 * - Check if the returned array is empty (404)
 * 
 * @async
 * @function getAllReservations
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * 
 * @returns {Array} 200 : Returns every reservation made on every catway (in a increasing order)
 * @returns {Error} 404 | 500
 */

exports.getAllReservations = async (req, res, next) => {

    try {

        const reservations = await Reservation.find().sort({catwayNumber : 1}); 

        if(reservations.length === 0){

            return res.status(404).json("Aucune réservation n'existe"); /* Avoids returning an empty array */

        }

        return res.status(200).json(reservations);

    } catch (error) {

        return res.status(500).json('Erreur serveur lors de la récupération des réservations');

    }
}

/**
 * GET ALL RESERVATIONS FOR A SPECIFIC CATWAY <br><br>
 * 
 * Get all the reservations associated to a catway by finding the catwayNumber <br><br>
 * 
 * - Check if the returned array is empty (404)
 * 
 * @async
 * @function getReservationsByCatway
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Array} Returns every reservation made on a specific catway
 * @returns {Error} 404 | 500
 */

exports.getReservationsByCatway = async (req, res, next) => {

    const id = req.params.id

    try {

        const reservations = await Reservation.find({catwayNumber : parseInt(id)});

        if(reservations.length === 0) {

            return res.status(404).json("Aucune réservation n'est associée à ce catway"); /* No empty array */

        }

        return res.status(200).json(reservations);


    } catch (error) {

        return res.status(500).json('Erreur serveur lors de la récupération des réservations associées à ce catway');

    }
}

/**
 * GET ONE SPECIFIC RESERVATION <br><br>
 * 
 * Get a reservation by its ID <br><br>
 * 
 * - Check if reservation exists (404)
 * 
 * @async
 * @function getReservationById
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Object} 200 : The reservation
 * @returns {Error} 404 | 500
 */

exports.getReservationById = async (req, res, next) => {

    const {id, idReservation} = req.params;

    try {

        const reservation = await Reservation.findOne({catwayNumber : parseInt(id), _id : idReservation});

        if(!reservation) {

            return res.status(404).json("Aucune réservation correspondante n'a été trouvée");

        } else {

            return res.status(200).json(reservation);

        }

    } catch (error) {

        return res.status(500).json("Cette réservation n'existe pas");

    }
}

/**
 * CREATE A RESERVATION <br><br>
 * 
 * - Check if mandatory fields are missing (client and boat names | starting and ending dates) (400) <br>
 * - Check if client and boat names are valid (both needs at least 3 characters) (400) <br>
 * - Check if end date isn't anterior to start date (400) <br>
 * - Check if the requested catway exists (404) <br>
 * - Check if there are any conflicts between booking dates and already existing bookings (409)
 * 
 * @async
 * @function createReservation
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Object} 201 : The created reservation
 * @returns {Error} 400 | 404 | 409 | 500
 */

exports.createReservation = async (req, res, next) => {

    const {clientName, boatName, startDate, endDate} = req.body;
    const {id} = req.params;

    try {

        if (!clientName || !boatName || !startDate || !endDate){

        return res.status(400).json('Un ou plusieurs champs sont manquants');

        }

        if (clientName.length < 3 || boatName.length < 3) {

        return res.status(400).json('Le nom du client et du navire doivent faire au moins 3 caractères');

        }

        if(endDate < startDate) {

        return res.status(400).json('La date de fin ne peut être antérieure à la date de début');

        }

        const catway = await Catway.findOne({catwayNumber : parseInt(id)});

        if(!catway) {

            return res.status(404).json("Ce catway n'existe pas");

        }

        const startingPoint = new Date(startDate);
        const endingPoint = new Date(endDate);

        const overlapping = await Reservation.find({
            catwayNumber : parseInt(id),
            startDate : {$lt : endingPoint}, /* Checks if reservation does not end after another reservation starts */
            endDate : {$gt : startingPoint} /* Checks if reservation does not start before another reservation ends */
        });

        if(overlapping.length > 0){

            return res.status(409).json('Créneau déjà réservé');
        }

        const newReservation = new Reservation ({
            catwayNumber : parseInt(id),
            clientName,
            boatName,
            startDate,
            endDate
        });

        await newReservation.save();

        return res.status(201).json(newReservation);

    } catch (error) {

        return res.status(500).json('Erreur serveur lors de la création de la réservation');
    }
}


/**
 * UPDATE A RESERVATION <br><br>
 * 
 * - Check if every required field is here (startDate and endDate) (400) <br>
 * - Check if endDate isn't anterior to startDate (400) <br>
 * - Check if clientName and boatName are being modified (400) <br>
 * - Check if reservation to update exists (404) <br>
 * - Check if new desired dates are not in direct conflict with existing reservations (the one to update excluded) (409)
 * 
 * @async
 * @function updateReservation
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Object} 201 : Updated reservation
 * @returns {Error} 400 | 404 | 409 | 500
 */

exports.updateReservation = async (req, res, next) => {

    const {startDate, endDate, clientName, boatName} = req.body;
    const {idReservation, id} = req.params;

    try {

        if(!startDate || !endDate){

            return res.status(400).json('Les nouvelles dates désirées sont requises');

        }

        if(endDate < startDate) {

            return res.status(400).json('La date de fin ne peut être antérieure à la date de début')

        }

        const updatedReservation = await Reservation.findOne({_id : idReservation});

        if(clientName !== updatedReservation.clientName || boatName !== updatedReservation.boatName) {

            return res.status(400).json('Les noms du client et du navire associés à la réservation ne peuvent être modifiés');

        }

        if(!updatedReservation) {

            return res.status(404).json("Cette réservation n'existe pas");

        } else {

            const newStartingDate = new Date(startDate);
            const newEndingDate = new Date(endDate);

            const overlapping = await Reservation.find({
                catwayNumber : parseInt(id),
                _id : {$ne : idReservation}, /* Excludes the reservation needing an update from overlapping checks */
                startDate : {$lt : newEndingDate}, /* Checks if the new ending date isn't after another reservation has started */
                endDate : {$gt : newStartingDate} /* Checks if the new starting date isn't before another reservation has ended */
            })

            if(overlapping.length > 0){

                return res.status(409).json('Créneau déjà utilisé');

            }

            updatedReservation.startDate = startDate;
            updatedReservation.endDate = endDate;

            await updatedReservation.save();

            return res.status(201).json(updatedReservation);
        }

    } catch (error) {

        return res.status(500).json('Erreur serveur lors de la mise à jour des dates de réservation');

    }
}

/**
 * DELETE A RESERVATION <br><br>
 * 
 * Use the ID of the reservation to find it <br><br>
 * 
 * -Check if reservation exists (404)
 * 
 * @async
 * @function deleteReservation
 * @param {Request} req Express request
 * @param {Response} res Express response
 * @param {NextFunction} next Callback to next function
 * 
 * @returns {Response} 200 : Successful deletion message
 * @returns {Error} 404 | 500
 */

exports.deleteReservation = async (req, res, next) => {

    const {idReservation} = req.params;

    try {

        const deletedReservation = await Reservation.findOne({_id : idReservation});

        if(!deletedReservation){

            return res.status(404).json("La réservation que vous souhaitez supprimer n'existe pas");

        } else {

            await deletedReservation.deleteOne();

            return res.status(200).json('Réservation supprimée'); 
            /** Using a 200 instead of a 204 allows a successful deletion message */

        }

    } catch (error) {

        return res.status(500).json('Erreur lors de la tentative de suppression');

    }
}
